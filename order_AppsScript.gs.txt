/**
 * 구글 시트 기록 + 텔레그램 알림 (order.html 전용 백엔드)
 *
 * 설정 방법
 * 1) 구글 시트를 새로 만듭니다.
 * 2) 메뉴에서 확장 프로그램 > Apps Script 를 클릭합니다.
 * 3) 기본으로 열려 있는 코드를 모두 지우고, 이 파일 내용을 전부 붙여넣습니다.
 * 4) 아래 TELEGRAM_BOT_TOKEN, TELEGRAM_CHAT_ID 값을 실제 값으로 채웁니다.
 *    - BOT_TOKEN: 텔레그램에서 @BotFather 검색 > /newbot > 안내에 따라 봇 이름 설정 > 발급되는 토큰 복사
 *    - CHAT_ID  : 만든 봇과 먼저 대화를 한 번 시작(아무 말이나 전송)한 뒤,
 *                 브라우저에서 https://api.telegram.org/bot<위에서 받은 토큰>/getUpdates 접속
 *                 → 응답 안의 "chat":{"id": 123456789 ...} 숫자를 복사
 * 5) 화면 오른쪽 위 [배포] > [새 배포] 클릭
 *    - 유형: 웹 앱
 *    - 실행 계정: 나(본인)
 *    - 액세스 권한: 모든 사용자
 *    - [배포] 클릭 후 나오는 웹 앱 URL을 복사
 * 6) order.html 파일 안의 ENDPOINT_URL 값에 그 URL을 붙여넣습니다.
 * 7) 이후 이 코드를 수정하면, [배포] > [배포 관리] > 연필 아이콘 > 새 버전으로
 *    다시 배포해야 변경사항이 실제 주소에 반영됩니다.
 */

var TELEGRAM_BOT_TOKEN = 'YOUR_BOT_TOKEN_HERE';
var TELEGRAM_CHAT_ID = 'YOUR_CHAT_ID_HERE';

function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var name = (data.name || '').toString().trim();
    var phone = (data.phone || '').toString().trim();
    var address = (data.address || '').toString().trim();
    var timestamp = new Date();

    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(['접수시각', '이름', '연락처', '택배주소']);
    }
    sheet.appendRow([timestamp, name, phone, address]);

    var message =
      '📚 새 주문 접수 — 인천부 혼마치 1943, 그들의 이름\n' +
      '이름: ' + name + '\n' +
      '연락처: ' + phone + '\n' +
      '주소: ' + address + '\n' +
      '시각: ' + Utilities.formatDate(timestamp, 'Asia/Seoul', 'yyyy-MM-dd HH:mm');

    if (TELEGRAM_BOT_TOKEN && TELEGRAM_BOT_TOKEN.indexOf('YOUR_BOT_TOKEN') !== 0) {
      UrlFetchApp.fetch('https://api.telegram.org/bot' + TELEGRAM_BOT_TOKEN + '/sendMessage', {
        method: 'post',
        contentType: 'application/json',
        payload: JSON.stringify({ chat_id: TELEGRAM_CHAT_ID, text: message }),
        muteHttpExceptions: true
      });
    }

    return ContentService.createTextOutput(JSON.stringify({ ok: true }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ ok: false, error: String(err) }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
